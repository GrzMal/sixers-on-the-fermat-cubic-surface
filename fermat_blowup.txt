proc genMap(list @F){ //ideal of the graph of the anticanonical map P2 -> P3
  ideal I=std(intersect(@F[1..6]));
  ideal J=I[1..4];
  matrix A[2][4]=J[1..4],x,y,z,w;
  if(system("version")<44100){
    ideal phi=std(sat(minor(A,2),J)[1]);
	}
  else{
	  ideal phi=std(sat(minor(A,2),J));
    }
  return(phi);
}
proc imPhi(ideal phi, ideal I){ //image of a subscheme of P2. For I=0 return the image surface
  ideal J=phi+I;
  if(system("version")<44100){
    J=std(eliminate(sat(J,ideal(a,b,c))[1],abc));
	}
  else{
	  J=std(eliminate(sat(J,ideal(a,b,c)),abc));
    }
  return(J);
}
proc imNaivePhi(list @F, ideal imId){ //image of a plane curve
  ideal I=std(intersect(@F[1..6]));
  ideal J=ideal(x-I[1],y-I[2],z-I[3],w-I[4])+imId;
  J=std(eliminate(J,abc));
  return(J);
}
proc imContrPhi(ideal @I, ideal imId){ //image of a plane curve under the second blow-down map P2 -> P2
  ideal J=std(eliminate(ideal(x-@I[1],y-@I[2],z-@I[3])+imId,abc));
  return(substitute(J,ideal(x,y,z),ideal(a,b,c)));
}
proc line(list pt){ //generate line
  ideal I=std(intersect(pt[1..2]));
  return(I[1]);
}
proc conic(list pt, int idx){ //equation of the conic through the five points different from pt[idx]
  ideal I=1;
	for(int i=1;i<=size(pt);i++){
	  if(i<>idx){
		I=intersect(I,pt[i]);
	  }
	}
  return(std(I)[1]);
}
proc idPoint(poly pp, poly qq, poly rr){ //ideal of given point
  matrix m[2][3]=pp,qq,rr,a,b,c;
  ideal I=minor(m,2);
  return(std(I));
}
proc eqIdeal(list L1, list L2){ //check if two ideals are equal
  ideal I=std(ideal(L1[1..size(L1)]));
  ideal J=std(ideal(L2[1..size(L2)]));
  if(size(NF(I,J))==0 && size(NF(J,I))==0){
	return(1);
  }
return(0);
}
proc combinations(int maxV, int nrPt){ //generates combinations
  if(nrPt>maxV){
  	return("n less than k");
  }
  for(int i=1;i<=nrPt;i++){
  	int t(i)=1;
  }
  string comm="for(t(1)=1;t(1)<=maxV-nrPt+1;t(1)++){";
  list AllCmb;
  for(i=2;i<=nrPt;i++){
  	comm=comm+"for(t("+string(i)+")=t("+string(i)+"-1)+1;t("+string(i)+")<=maxV-nrPt+"+string(i)+";t("+string(i)+")++){";
  }
  comm=comm+"list Cmb; for(i=1;i<=nrPt;i++){Cmb=Cmb+list(t(i));} AllCmb[size(AllCmb)+1]=Cmb; kill Cmb;";
  for(i=1;i<=nrPt;i++){
    comm=comm+"}";
  }
  execute(comm);
return(AllCmb);	
}
proc Lidx(int i, int j){ //index for L_ij
  if(i<j){
	return(10*i+j);
  }
return(10*j+i);
}
proc isIn(int v, list A){ //checks whether v is in A
  for(int i=1;i<=size(A);i++){
    if(v==A[i]){
      return(1);
    }
  }
return(0);
}
proc compl6(list A){ //complement of set in {1,...,6}
  list B;
  for(int i=1;i<=6;i++){
    if(isIn(i,A)==0){
      B[size(B)+1]=i;
    }
  }
return(B);
}
proc genSixers(list E, list L, list Q){ //generate all 72 sixers
  int i,j,k; list sixers;
  sixers[1]=E; //type E: E_1,...,E_6
  sixers[2]=Q; //type Q: Q_1,...,Q_6
  list comb63=combinations(6,3);
  for(int s=1;s<=size(comb63);s++){ //type 3E+3L: E_i, E_j, E_k and L_ab
    list I=comb63[s];
    list J=compl6(I);
    list S;
    S[1]=E[I[1]];
    S[2]=E[I[2]];
    S[3]=E[I[3]];
    S[4]=L[Lidx(J[1],J[2])];
    S[5]=L[Lidx(J[1],J[3])];
    S[6]=L[Lidx(J[2],J[3])];
    sixers[size(sixers)+1]=S;
    kill S,I,J;
  }
  for(s=1;s<=size(comb63);s++) { //type 3L+3Q: L_ab, and Q_i, Q_j, Q_k
    list I=comb63[s];
    list J=compl6(I);
    list S;
    S[1]=L[Lidx(I[1],I[2])];
    S[2]=L[Lidx(I[1],I[3])];
    S[3]=L[Lidx(I[2],I[3])];
    S[4]=Q[J[1]];
    S[5]=Q[J[2]];
    S[6]=Q[J[3]];
    sixers[size(sixers)+1]=S;
    kill S,I,J;
  }
  for(i=1;i<=6;i++){ //type E+Q+4L: {E_i,Q_i} + {L_kj: j not in {i,k}}
  	for(k=1;k<=6;k++){
  	  if (i<>k){
  	    list S;
  	    S[1]=E[i];
  	    S[2]=Q[i];
  	    int pos=3;
        for(j=1;j<=6;j++){
  		  if(j<>i){
  		    if(j<>k){
  			  S[pos]=L[Lidx(k,j)];
  			  pos++;
  		    }
  		  }
        }
  	  sixers[size(sixers)+1]=S;
  	  kill S,pos;
  	  }
  	}
  }
return(sixers);
}
proc coordChange(list @L, matrix @m){ //pull back an ideal in P3 by the projective transformation defined by @m
  matrix @v[4][1]=x,y,z,w;
  for(int i=1;i<=size(@L[1]);i++){
	@L[1][i]=std(substitute(@L[1][i],ideal(x,y,z,w),ideal(@m*@v)))[1];
  }
return(@L[1]);
}
